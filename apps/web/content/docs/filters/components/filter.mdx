## Installation

```npm
npx shadcn@latest add {{REGISTRY_URL}}/r/filters{{RELEASE_CHANNEL}}
```

## Usage

```tsx
import { Filter } from '@/components/ui/filter'
```

```tsx
<Filter.Provider
  columns={columns}
  filters={filters}
  actions={actions}
  strategy={strategy}
  locale="en"
>
  <Filter.Root>
    <Filter.Menu />
    <Filter.List>
      {({ filter, column }) => (
        <Filter.Item filter={filter} column={column}>
          <Filter.Subject />
          <Filter.Operator />
          <Filter.Value />
          <Filter.Remove />
        </Filter.Item>
      )}
    </Filter.List>
    <Filter.Actions />
  </Filter.Root>
</Filter.Provider>
```

<Filter.Provider
  columns={columns}

### Custom trigger for filter menu

Use the `asChild` prop on `Filter.Trigger` to render a custom trigger element:

```tsx
<Filter.Menu>
  <Filter.Trigger asChild>
    <Button variant="outline" size="sm">
      <PlusIcon className="size-4 mr-2" />
      Add Filter
    </Button>
  </Filter.Trigger>
</Filter.Menu>
```

### Mobile Layout

Use `Filter.ListMobileContainer` for a horizontally scrollable list with blur effects on mobile:

```tsx
<Filter.Root>
  <Filter.Menu />
  <Filter.ListMobileContainer>
    <Filter.List>
      {({ filter, column }) => (
        <Filter.Item filter={filter} column={column}>
          <Filter.Subject />
          <Filter.Operator />
          <Filter.Value />
          <Filter.Remove />
        </Filter.Item>
      )}
    </Filter.List>
  </Filter.ListMobileContainer>
  <Filter.Actions />
</Filter.Root>
```

### Standalone Item Components

Filter item sub-components can be used standalone with explicit props (outside of `Filter.Item` context):

```tsx
// Using Filter.Subject without FilterItem context
<Filter.Subject column={myColumn} entityName="Issue" />

// Using Filter.Operator without FilterItem context
<Filter.Operator
  column={myColumn}
  filter={myFilter}
  actions={actions}
  locale="en"
/>
```

### Using Context Hooks

Access filter context anywhere within the `Filter.Provider`:

```tsx
import { Filter } from '@/registry/ui/filter'

function CustomFilterDisplay() {
  const { filters, columns } = Filter.useFilterContext()
  const actions = Filter.useFilterActions()
  const locale = Filter.useFilterLocale()

  return (
    <div>
      <span>Active filters: {filters.length}</span>
      <button onClick={() => actions.removeAllFilters()}>
        Clear All
      </button>
    </div>
  )
}
```

## Context Hooks

These hooks can be used anywhere within a `Filter.Provider` to access filter state and actions.

### useFilterContext

Returns the full filter context value.

```tsx
const { columns, filters, actions, strategy, locale, entityName, variant } =
  Filter.useFilterContext<TData>()
```

### useFilterActions

Returns the filter actions object.

```tsx
const actions = Filter.useFilterActions()

actions.setFilterValue(column, values)
actions.setFilterOperator(columnId, operator)
actions.addFilterValue(column, values)
actions.removeFilterValue(column, values)
actions.removeFilter(columnId)
actions.removeAllFilters()
```

### useFilterColumn

Returns a specific column configuration by ID.

```tsx
const column = Filter.useFilterColumn<TData>(columnId)
```

### useFilterStrategy

Returns the current filter strategy.

```tsx
const strategy = Filter.useFilterStrategy() // 'client' | 'server'
```

### useFilterLocale

Returns the current locale.

```tsx
const locale = Filter.useFilterLocale() // 'en' | 'de' | ...
```

### useFilterEntityName

Returns the entity name (used for boolean filter display).

```tsx
const entityName = Filter.useFilterEntityName()
```

### useFilterItemContext

Returns the FilterItem context (only available inside `Filter.Item`). Returns `null` if not inside a FilterItem.

```tsx
const itemContext = Filter.useFilterItemContext<TData, TType>()
// { filter, column, actions, strategy, locale, entityName }
```

## Advanced: Value Utilities

For advanced customization of filter value menus, the following utilities are exported:

### Menu Creators

```tsx
import {
  createOptionMenu,
  createMultiOptionMenu,
  createTextMenu,
  createTextFilterMiddleware
} from '@/registry/ui/filter'

// Create option menu configuration
const optionMenu = createOptionMenu({
  column: myOptionColumn,
  actions,
  filter: currentFilter,
  strategy,
  locale,
})

// Create multi-option menu configuration
const multiOptionMenu = createMultiOptionMenu({
  column: myMultiOptionColumn,
  actions,
  filter: currentFilter,
  strategy,
  locale,
})

// Create text menu configuration
const textMenu = createTextMenu({
  column: myTextColumn,
  actions,
  strategy,
  locale,
})
```

### Value Controllers

For custom implementations, you can use the value controllers directly:

```tsx
import {
  FilterValueDateController,
  FilterValueNumberController,
} from '@/registry/ui/filter'

// Date controller with calendar
<FilterValueDateController
  filter={dateFilter}
  column={dateColumn}
  actions={actions}
  strategy={strategy}
  locale="en"
/>

// Number controller with slider and inputs
<FilterValueNumberController
  filter={numberFilter}
  column={numberColumn}
  actions={actions}
  strategy={strategy}
  locale="en"
/>
```

## TypeScript

The Filter components are fully typed. Use generics to get proper typing for your data:

```tsx
import type {
  FilterProviderProps,
  FilterItemProps,
  FilterContextValue,
  FilterItemContextValue,
} from '@/registry/ui/filter'

// Type your filter provider
<Filter.Provider<MyDataType>
  columns={columns}
  filters={filters}
  actions={actions}
  strategy={strategy}
>
  {/* ... */}
</Filter.Provider>

// Type your filter items
<Filter.Item<MyDataType, 'option'>
  filter={optionFilter}
  column={optionColumn}
>
  {/* ... */}
</Filter.Item>
```

# Navigation Callback

The `onNavigationChange` callback fires whenever a user navigates into a submenu (forward) or returns to a parent menu (back). This enables animations, analytics tracking, custom UI updates, and more.

## Overview

Every time the navigation stack changes, `onNavigationChange` is called with a `NavigationChangeEvent` containing:

- **direction** - Whether navigation is `'forward'` (entering submenu) or `'back'` (returning to parent)
- **prevBreadcrumbs** - The navigation stack before the change
- **nextBreadcrumbs** - The navigation stack after the change

## Basic Usage

```tsx
import { CommandMenu, type NavigationChangeEvent } from '@bazza-ui/command-menu'

function App() {
  const handleNavigationChange = (event: NavigationChangeEvent) => {
    console.log('Navigation direction:', event.direction)
    console.log('Previous breadcrumbs:', event.prevBreadcrumbs)
    console.log('Next breadcrumbs:', event.nextBreadcrumbs)
  }

  return (
    <CommandMenu
      menu={menu}
      onNavigationChange={handleNavigationChange}
    />
  )
}
```

## NavigationChangeEvent Type

```typescript
type NavigationChangeEvent = {
  /** Direction of navigation */
  direction: 'forward' | 'back'
  /** Previous navigation stack state */
  prevBreadcrumbs: NavigationStackEntry[]
  /** New navigation stack state */
  nextBreadcrumbs: NavigationStackEntry[]
}

type NavigationStackEntry = {
  menuId: string
  menuTitle?: string
  parentMenuId?: string
}
```

## Use Cases

### 1. Animations

Trigger different animations based on navigation direction:

```tsx
function AnimatedMenu() {
  const [animClass, setAnimClass] = useState('')

  const handleNavigationChange = ({ direction }: NavigationChangeEvent) => {
    if (direction === 'forward') {
      setAnimClass('slide-left')
    } else {
      setAnimClass('slide-right')
    }

    // Reset after animation completes
    setTimeout(() => setAnimClass(''), 300)
  }

  return (
    <div className={animClass}>
      <CommandMenu
        menu={menu}
        onNavigationChange={handleNavigationChange}
      />
    </div>
  )
}
```

See [Dialog Animations](/docs/command-menu/dialog-animations) for complete animation guide.

### 2. Analytics Tracking

Track how users navigate through your command menu:

```tsx
function TrackedMenu() {
  const handleNavigationChange = (event: NavigationChangeEvent) => {
    // Track navigation depth
    analytics.track('command_menu_navigation', {
      direction: event.direction,
      depth: event.nextBreadcrumbs.length,
      from: event.prevBreadcrumbs[event.prevBreadcrumbs.length - 1]?.menuId,
      to: event.nextBreadcrumbs[event.nextBreadcrumbs.length - 1]?.menuId,
    })
  }

  return <CommandMenu menu={menu} onNavigationChange={handleNavigationChange} />
}
```

### 3. Custom Breadcrumbs

Build a custom breadcrumb UI outside the menu:

```tsx
function MenuWithExternalBreadcrumbs() {
  const [breadcrumbs, setBreadcrumbs] = useState<NavigationStackEntry[]>([])

  const handleNavigationChange = ({ nextBreadcrumbs }: NavigationChangeEvent) => {
    setBreadcrumbs(nextBreadcrumbs)
  }

  return (
    <>
      {/* External breadcrumb UI */}
      <div className="breadcrumbs">
        {breadcrumbs.map((entry, i) => (
          <span key={entry.menuId}>
            {i > 0 && ' / '}
            {entry.menuTitle || entry.menuId}
          </span>
        ))}
      </div>

      <CommandMenu
        menu={menu}
        showBreadcrumbs={false} // Hide internal breadcrumbs
        onNavigationChange={handleNavigationChange}
      />
    </>
  )
}
```

### 4. Navigation Depth Limit

Prevent navigation beyond a certain depth:

```tsx
function LimitedDepthMenu() {
  const MAX_DEPTH = 3

  const handleNavigationChange = ({ direction, nextBreadcrumbs }: NavigationChangeEvent) => {
    if (direction === 'forward' && nextBreadcrumbs.length >= MAX_DEPTH) {
      toast.warning(`Maximum depth of ${MAX_DEPTH} levels reached`)
    }
  }

  return <CommandMenu menu={menu} onNavigationChange={handleNavigationChange} />
}
```

### 5. State Synchronization

Sync navigation state with URL or external state:

```tsx
function StateSyncedMenu() {
  const [searchParams, setSearchParams] = useSearchParams()

  const handleNavigationChange = ({ nextBreadcrumbs }: NavigationChangeEvent) => {
    // Sync breadcrumbs to URL
    const path = nextBreadcrumbs.map(b => b.menuId).join('/')
    setSearchParams({ menu: path })
  }

  return <CommandMenu menu={menu} onNavigationChange={handleNavigationChange} />
}
```

### 6. Conditional UI Updates

Show/hide UI elements based on navigation depth:

```tsx
function ConditionalUIMenu() {
  const [showHelpText, setShowHelpText] = useState(true)

  const handleNavigationChange = ({ nextBreadcrumbs }: NavigationChangeEvent) => {
    // Hide help text when in submenus
    setShowHelpText(nextBreadcrumbs.length === 0)
  }

  return (
    <>
      {showHelpText && <HelpText />}
      <CommandMenu menu={menu} onNavigationChange={handleNavigationChange} />
    </>
  )
}
```

### 7. Performance Monitoring

Track navigation performance:

```tsx
function MonitoredMenu() {
  const navigationStartTime = useRef<number>()

  const handleNavigationChange = ({ direction }: NavigationChangeEvent) => {
    if (navigationStartTime.current) {
      const duration = performance.now() - navigationStartTime.current
      console.log(`Navigation took ${duration}ms`)
    }
    navigationStartTime.current = performance.now()
  }

  return <CommandMenu menu={menu} onNavigationChange={handleNavigationChange} />
}
```

## Understanding Breadcrumbs

The breadcrumbs arrays contain the full navigation path:

```tsx
// Root level (no submenus)
prevBreadcrumbs: []
nextBreadcrumbs: []

// First submenu entered
prevBreadcrumbs: []
nextBreadcrumbs: [
  { menuId: 'settings', menuTitle: 'Settings' }
]

// Second submenu entered
prevBreadcrumbs: [
  { menuId: 'settings', menuTitle: 'Settings' }
]
nextBreadcrumbs: [
  { menuId: 'settings', menuTitle: 'Settings' },
  { menuId: 'advanced', menuTitle: 'Advanced Settings' }
]

// Navigating back
prevBreadcrumbs: [
  { menuId: 'settings', menuTitle: 'Settings' },
  { menuId: 'advanced', menuTitle: 'Advanced Settings' }
]
nextBreadcrumbs: [
  { menuId: 'settings', menuTitle: 'Settings' }
]
```

## Direction Values

```typescript
// User enters a submenu (clicks submenu item or presses Enter/Right arrow)
direction: 'forward'

// User returns to parent (clicks back button, presses Escape/Left arrow)
direction: 'back'
```

## Timing

The callback fires **synchronously** during the state update, before the new menu is rendered. This ensures you can:

- Update state immediately for animations
- Prevent race conditions
- Capture exact before/after state

```tsx
const handleNavigationChange = (event: NavigationChangeEvent) => {
  // ✅ This runs BEFORE the new menu renders
  console.log('About to navigate')
  setAnimationState(event.direction)

  // ❌ Don't rely on async operations here
  // The menu will render before this completes
  setTimeout(() => {
    setAnimationState('idle')
  }, 1000)
}
```

## Combining with DialogInner Slot

The most powerful pattern is combining `onNavigationChange` with the `DialogInner` slot:

```tsx
function AnimatedCommandMenu() {
  const [direction, setDirection] = useState<'forward' | 'back' | 'idle'>('idle')

  return (
    <CommandMenu
      menu={menu}
      // 1. Capture navigation direction
      onNavigationChange={({ direction }) => {
        setDirection(direction)
        setTimeout(() => setDirection('idle'), 300)
      }}
      theme={{
        slots: {
          // 2. Apply animation based on direction
          DialogInner: ({ children, baseProps }) => (
            <div
              {...baseProps}
              className={cn(
                baseProps.className,
                direction === 'forward' && 'animate-slide-left',
                direction === 'back' && 'animate-slide-right',
              )}
            >
              {children}
            </div>
          ),
        },
      }}
    />
  )
}
```

## TypeScript

The callback is fully typed:

```typescript
import type { NavigationChangeEvent } from '@bazza-ui/command-menu'

// Inline
<CommandMenu
  onNavigationChange={(event: NavigationChangeEvent) => {
    // event.direction is 'forward' | 'back'
    // event.prevBreadcrumbs is NavigationStackEntry[]
    // event.nextBreadcrumbs is NavigationStackEntry[]
  }}
/>

// External function
function handleNavigation(event: NavigationChangeEvent) {
  // Fully typed!
}

<CommandMenu onNavigationChange={handleNavigation} />
```

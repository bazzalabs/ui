# Dialog Animations

Add custom animations when navigating between submenus using the `DialogInner` slot combined with the `onNavigationChange` callback.

## Overview

The command menu provides two primitives for implementing submenu navigation animations:

1. **`onNavigationChange` callback** - Fires when navigating forward/back, provides direction and breadcrumb state
2. **`DialogInner` slot** - Customizes the inner dialog wrapper where you can apply animation classes

Together, these allow you to implement smooth transitions like slides, fades, or custom animations when users navigate between menus.

## Basic Example

Here's a simple slide animation using CSS:

```tsx
'use client'

import { useState, useRef } from 'react'
import { CommandMenu, type NavigationChangeEvent } from '@bazza-ui/command-menu'
import { cn } from '@/lib/utils'

export function AnimatedCommandMenu() {
  const [navDirection, setNavDirection] = useState<'forward' | 'back' | 'idle'>('idle')
  const timerRef = useRef<NodeJS.Timeout>()

  const handleNavigationChange = (event: NavigationChangeEvent) => {
    // Set direction for animation
    setNavDirection(event.direction)

    // Reset to idle after animation completes (300ms)
    clearTimeout(timerRef.current)
    timerRef.current = setTimeout(() => {
      setNavDirection('idle')
    }, 300)
  }

  return (
    <CommandMenu
      menu={menu}
      onNavigationChange={handleNavigationChange}
      theme={{
        slots: {
          DialogInner: ({ children, baseProps }) => (
            <div
              {...baseProps}
              className={cn(
                baseProps.className,
                navDirection === 'forward' && 'animate-slide-left',
                navDirection === 'back' && 'animate-slide-right',
              )}
            >
              {children}
            </div>
          ),
        },
      }}
    />
  )
}
```

Add these animations to your CSS:

```css
@keyframes slide-left {
  from {
    transform: translateX(100%);
    opacity: 0;
  }
  to {
    transform: translateX(0);
    opacity: 1;
  }
}

@keyframes slide-right {
  from {
    transform: translateX(-100%);
    opacity: 0;
  }
  to {
    transform: translateX(0);
    opacity: 1;
  }
}

.animate-slide-left {
  animation: slide-left 300ms ease-out;
}

.animate-slide-right {
  animation: slide-right 300ms ease-out;
}
```

## The DialogInner Slot

The `DialogInner` slot replaces the default inner wrapper element of the command menu dialog.

### Slot Signature

```typescript
DialogInner?: (props: {
  children: React.ReactNode
  baseProps: {
    ref: React.RefCallback<HTMLDivElement>
    className?: string
    'data-slot': 'command-menu-dialog-inner'
    'data-command-menu-dialog-inner': true
  }
}) => React.ReactElement
```

### Important: baseProps Must Be Applied

The `baseProps` **must** be spread onto your wrapper element. It contains a ref callback that powers the ResizeObserver for smooth height transitions.

```tsx
// ✅ Correct - baseProps applied
DialogInner: ({ children, baseProps }) => (
  <div {...baseProps} className={cn(baseProps.className, 'custom-class')}>
    {children}
  </div>
)

// ❌ Wrong - missing baseProps
DialogInner: ({ children }) => (
  <div className="custom-class">
    {children}
  </div>
)
```

## Animation Examples

### Slide Transitions

Slide left when entering a submenu, slide right when going back:

```tsx
theme={{
  slots: {
    DialogInner: ({ children, baseProps }) => {
      const [direction, setDirection] = useState<'forward' | 'back' | 'idle'>('idle')

      return (
        <div
          {...baseProps}
          className={cn(
            baseProps.className,
            direction === 'forward' && 'animate-in slide-in-from-right duration-200',
            direction === 'back' && 'animate-in slide-in-from-left duration-200',
          )}
        >
          {children}
        </div>
      )
    },
  },
}}
```

### Fade Transitions

Simple cross-fade between menus:

```tsx
theme={{
  slots: {
    DialogInner: ({ children, baseProps }) => (
      <div
        {...baseProps}
        className={cn(
          baseProps.className,
          'animate-in fade-in duration-150',
        )}
      >
        {children}
      </div>
    ),
  },
}}
```

### Slide + Fade Combo

Combine slide and fade for a polished transition:

```tsx
theme={{
  slots: {
    DialogInner: ({ children, baseProps }) => {
      const [direction, setDirection] = useState<'forward' | 'back' | 'idle'>('idle')

      return (
        <div
          {...baseProps}
          className={cn(
            baseProps.className,
            'transition-all duration-200',
            direction === 'forward' && 'animate-in slide-in-from-right-4 fade-in',
            direction === 'back' && 'animate-in slide-in-from-left-4 fade-in',
          )}
        >
          {children}
        </div>
      )
    },
  },
}}
```

## Using Framer Motion

For more complex animations, use Framer Motion:

```tsx
import { motion, AnimatePresence } from 'framer-motion'

export function AnimatedCommandMenu() {
  const [navDirection, setNavDirection] = useState<'forward' | 'back'>('forward')
  const [key, setKey] = useState(0)

  const handleNavigationChange = (event: NavigationChangeEvent) => {
    setNavDirection(event.direction)
    setKey(prev => prev + 1) // Force re-mount for AnimatePresence
  }

  const variants = {
    enter: (direction: 'forward' | 'back') => ({
      x: direction === 'forward' ? 300 : -300,
      opacity: 0,
    }),
    center: {
      x: 0,
      opacity: 1,
    },
    exit: (direction: 'forward' | 'back') => ({
      x: direction === 'forward' ? -300 : 300,
      opacity: 0,
    }),
  }

  return (
    <CommandMenu
      menu={menu}
      onNavigationChange={handleNavigationChange}
      theme={{
        slots: {
          DialogInner: ({ children, baseProps }) => (
            <div {...baseProps}>
              <AnimatePresence initial={false} custom={navDirection} mode="wait">
                <motion.div
                  key={key}
                  custom={navDirection}
                  variants={variants}
                  initial="enter"
                  animate="center"
                  exit="exit"
                  transition={{
                    x: { type: "spring", stiffness: 300, damping: 30 },
                    opacity: { duration: 0.2 },
                  }}
                >
                  {children}
                </motion.div>
              </AnimatePresence>
            </div>
          ),
        },
      }}
    />
  )
}
```

## Pure CSS Approach

For a dependency-free solution, use CSS with data attributes:

```tsx
export function CSSAnimatedCommandMenu() {
  const [navDirection, setNavDirection] = useState<'idle' | 'forward' | 'back'>('idle')

  return (
    <div data-nav-direction={navDirection}>
      <CommandMenu
        menu={menu}
        onNavigationChange={({ direction }) => {
          setNavDirection(direction)
          setTimeout(() => setNavDirection('idle'), 300)
        }}
      />
    </div>
  )
}
```

Then target in your CSS:

```css
[data-nav-direction="forward"] [data-command-menu-dialog-inner] {
  animation: slide-left 300ms ease-out;
}

[data-nav-direction="back"] [data-command-menu-dialog-inner] {
  animation: slide-right 300ms ease-out;
}

[data-nav-direction="idle"] [data-command-menu-dialog-inner] {
  animation: none;
}
```

## Factory Pattern

For design systems, set animations at the factory level:

```tsx
import { createCommandMenu } from '@bazza-ui/command-menu'

const { CommandMenu, Root, Trigger, Content } = createCommandMenu({
  slots: {
    DialogInner: ({ children, baseProps }) => {
      // Your animation logic here
      return (
        <div {...baseProps} className={cn(baseProps.className, 'animate-transition')}>
          {children}
        </div>
      )
    },
  },
})

// Now all command menus in your app have animations by default
export { CommandMenu }
```

## Performance Tips

1. **Use CSS animations when possible** - More performant than JS-based animations
2. **Keep animations short** - 150-300ms is ideal for navigation
3. **Use transform and opacity** - These properties are GPU-accelerated
4. **Avoid animating height** - The dialog already handles height transitions
5. **Clean up timers** - Always clear timeouts to prevent memory leaks

## Common Patterns

### State Management

```tsx
// Using useState
const [direction, setDirection] = useState<'forward' | 'back' | 'idle'>('idle')

// Using useRef (no re-renders)
const directionRef = useRef<'forward' | 'back' | 'idle'>('idle')

// Using CSS variables
const contentRef = useRef<HTMLDivElement>(null)
useEffect(() => {
  contentRef.current?.style.setProperty('--nav-direction', direction)
}, [direction])
```

### Animation Timing

```tsx
const ANIMATION_DURATION = 300

const handleNavigationChange = (event: NavigationChangeEvent) => {
  setDirection(event.direction)
  setTimeout(() => setDirection('idle'), ANIMATION_DURATION)
}
```

### Conditional Animations

Only animate for specific navigation levels:

```tsx
const handleNavigationChange = (event: NavigationChangeEvent) => {
  // Only animate if going deeper than 2 levels
  if (event.nextBreadcrumbs.length > 1) {
    setDirection(event.direction)
  }
}
```

A combobox combines a text input with a dropdown list of suggestions. It's ideal for autocomplete scenarios, searchable selects, and tagging interfaces. Supports both single-value and multi-value selection modes.

## Installation

```bash
bun add @bazza-ui/react
```

## Basic Usage

```tsx
import { Combobox } from '@bazza-ui/react'

function Demo() {
  const [value, setValue] = React.useState('')

  return (
    <Combobox.Root value={value} onValueChange={setValue}>
      <Combobox.InputWrapper>
        <Combobox.Input placeholder="Search fruits..." />
        <Combobox.Clear />
      </Combobox.InputWrapper>
      <Combobox.Portal>
        <Combobox.Positioner>
          <Combobox.Popup>
            <Combobox.Item value="apple">
              <Combobox.ItemIndicator />
              <Combobox.ItemLabel>Apple</Combobox.ItemLabel>
            </Combobox.Item>
            <Combobox.Item value="banana">
              <Combobox.ItemIndicator />
              <Combobox.ItemLabel>Banana</Combobox.ItemLabel>
            </Combobox.Item>
            <Combobox.Item value="orange">
              <Combobox.ItemIndicator />
              <Combobox.ItemLabel>Orange</Combobox.ItemLabel>
            </Combobox.Item>
            <Combobox.Empty>No fruits found</Combobox.Empty>
          </Combobox.Popup>
        </Combobox.Positioner>
      </Combobox.Portal>
    </Combobox.Root>
  )
}
```

## Features

- **Autocomplete** - Filter suggestions as user types
- **Single & Multi-select** - Support for selecting one or multiple values
- **Free-form input** - Allow users to enter custom values not in the list
- **Fuzzy matching** - Built-in fuzzy search algorithm
- **Groups** - Organize options into labeled groups
- **Virtualized** - Support for virtualized lists with thousands of items
- **Controlled & Uncontrolled** - Works in both modes

## Examples

### Multi-Select Tags

```tsx
function TagsCombobox() {
  const [values, setValues] = React.useState<string[]>([])

  return (
    <Combobox.Root
      value={values}
      onValueChange={setValues}
      multiple
    >
      <Combobox.InputWrapper>
        <div className="flex flex-wrap gap-1">
          {values.map((tag) => (
            <span key={tag} className="px-2 py-1 bg-blue-100 rounded text-sm">
              {tag}
              <button onClick={() => setValues(values.filter((v) => v !== tag))}>
                x
              </button>
            </span>
          ))}
          <Combobox.Input placeholder="Add tags..." />
        </div>
        <Combobox.Clear />
      </Combobox.InputWrapper>
      <Combobox.Portal>
        <Combobox.Positioner>
          <Combobox.Popup>
            <Combobox.Item value="react">React</Combobox.Item>
            <Combobox.Item value="typescript">TypeScript</Combobox.Item>
            <Combobox.Item value="javascript">JavaScript</Combobox.Item>
            <Combobox.Item value="css">CSS</Combobox.Item>
            <Combobox.Item value="html">HTML</Combobox.Item>
          </Combobox.Popup>
        </Combobox.Positioner>
      </Combobox.Portal>
    </Combobox.Root>
  )
}
```

### With Custom Filtering

```tsx
function CustomFilterCombobox() {
  const customFilter = (value: string, search: string) => {
    // Case-insensitive starts-with matching
    return value.toLowerCase().startsWith(search.toLowerCase()) ? 1 : 0
  }

  return (
    <Combobox.Root filter={customFilter}>
      <Combobox.InputWrapper>
        <Combobox.Input placeholder="Type to filter..." />
      </Combobox.InputWrapper>
      <Combobox.Portal>
        <Combobox.Positioner>
          <Combobox.Popup>
            <Combobox.Item value="apple">Apple</Combobox.Item>
            <Combobox.Item value="apricot">Apricot</Combobox.Item>
            <Combobox.Item value="banana">Banana</Combobox.Item>
            <Combobox.Item value="blueberry">Blueberry</Combobox.Item>
          </Combobox.Popup>
        </Combobox.Positioner>
      </Combobox.Portal>
    </Combobox.Root>
  )
}
```

### Allow Custom Values

```tsx
function FreetextCombobox() {
  const [value, setValue] = React.useState('')

  return (
    <Combobox.Root
      value={value}
      onValueChange={setValue}
      allowCustomValue
    >
      <Combobox.InputWrapper>
        <Combobox.Input placeholder="Type or select..." />
      </Combobox.InputWrapper>
      <Combobox.Portal>
        <Combobox.Positioner>
          <Combobox.Popup>
            <Combobox.Item value="option1">Option 1</Combobox.Item>
            <Combobox.Item value="option2">Option 2</Combobox.Item>
            <Combobox.Item value="option3">Option 3</Combobox.Item>
            <Combobox.Empty>
              Press Enter to add "{value}"
            </Combobox.Empty>
          </Combobox.Popup>
        </Combobox.Positioner>
      </Combobox.Portal>
    </Combobox.Root>
  )
}
```

## Differences from Select

| Feature | Combobox | Select |
|---------|----------|--------|
| Input | Text input for typing | Button trigger |
| Filtering | As-you-type filtering | Optional search input |
| Custom values | Can allow free-form input | No custom values |
| Multi-select | Native support | Not supported |
| Use case | Autocomplete, tagging | Single selection |
